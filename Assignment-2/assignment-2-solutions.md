## Question 1

Create table `Member` and add constraints using ALTER.

### SQL Query

```sql
CREATE TABLE MEMBER (
    MEMBER_ID NUMBER(5),
    MEMBER_NAME VARCHAR2(30),
    MEMBER_ADDRESS VARCHAR2(50),
    ACC_OPEN_DATE DATE,
    MEMBERSHIP_TYPE VARCHAR2(20),
    FEES_PAID NUMBER(4),
    MAX_BOOKS_ALLOWED NUMBER(2),
    PENALTY_AMOUNT NUMBER(6)
);
```

### Add Constraints

```sql
ALTER TABLE MEMBER ADD PRIMARY KEY (MEMBER_ID);

ALTER TABLE MEMBER MODIFY MEMBER_NAME VARCHAR2(30) NOT NULL;

ALTER TABLE MEMBER 
ADD CHECK (MEMBERSHIP_TYPE IN ('LIFETIME','ANNUAL','HALF YEARLY','QUARTERLY'));

ALTER TABLE MEMBER 
ADD CHECK (MAX_BOOKS_ALLOWED < 7);

ALTER TABLE MEMBER 
ADD CHECK (PENALTY_AMOUNT <= 1000);
```

## Question 2

Create table `Books` and add constraints using ALTER.

### SQL Query

```sql
CREATE TABLE BOOKS (
    BOOK_NO NUMBER(6),
    BOOK_NAME VARCHAR2(30),
    AUTHOR_NAME VARCHAR2(30),
    COST NUMBER(7,2),
    CATEGORY CHAR(10)
);
```

### Add Constraints

```sql
ALTER TABLE BOOKS ADD PRIMARY KEY (BOOK_NO);

ALTER TABLE BOOKS MODIFY BOOK_NAME VARCHAR2(30) NOT NULL;

ALTER TABLE BOOKS 
ADD CHECK (CATEGORY IN ('SCIENCE','DATABASE','SYSTEM','OTHERS'));
```

## Question 3

Create table `Issue` and add constraints using ALTER.

### SQL Query

```sql
CREATE TABLE ISSUE (
    LIB_ISSUE_ID NUMBER(10),
    BOOK_NO NUMBER(6),
    MEMBER_ID NUMBER(5),
    ISSUE_DATE DATE,
    RETURN_DATE DATE
);
```

### Add Constraints

```sql
ALTER TABLE ISSUE ADD PRIMARY KEY (LIB_ISSUE_ID);

ALTER TABLE ISSUE 
ADD FOREIGN KEY (BOOK_NO) REFERENCES BOOKS(BOOK_NO);

ALTER TABLE ISSUE 
ADD FOREIGN KEY (MEMBER_ID) REFERENCES MEMBER(MEMBER_ID);
```
